Alles, was die On-Ramp an Syntax braucht — auf einen Blick, mit Pinguin- und Betriebsdaten-Beispielen.

Grundgerüst ggplot2 (Lektion 0001)

ggplot(penguins, aes(x = bill_length_mm, y = bill_depth_mm)) +
  geom_point()
# Daten in ggplot(), Rollen in aes(), Darstellung als Geom,
#  Schichten mit + — das + steht IMMER am Zeilenende

Geoms nach Frage (Lektion 0001 + 0003)

ggplot(penguins, aes(x = species, y = body_mass_g)) +
  geom_boxplot()                       # Verteilung je Gruppe

ggplot(fertigung, aes(x = masse_mm)) +
  geom_histogram(binwidth = 0.05) +    # binwidth probieren!
  geom_vline(xintercept = 10, linetype = "dashed")  # Sollwert

ggplot(penguins, aes(x = bill_length_mm, y = bill_depth_mm)) +
  geom_point() +
  facet_wrap(~ species)                # trennen statt färben

Die vier dplyr-Verben (Lektion 0002)

penguins |> filter(species == "Gentoo")        # Zeilen wählen (== ! =)
penguins |> mutate(kg = body_mass_g / 1000)    # Spalte umformen/neu
penguins |> group_by(species)                  # Gruppen bilden
penguins |>
  group_by(species) |>
  summarise(gram = mean(body_mass_g, na.rm = TRUE))
# pro Gruppe eine Kennzahl — ohne na.rm: NA!

Einlesen + Kennzahlen (Lektion 0003)

fertigung <- read_csv("unterricht/HWIT-PMM/assets/betriebsdaten.csv")
glimpse(fertigung)     # Zeilen, Spalten, Typen — der Pflicht-Check
summary(fertigung$masse_mm)  # Min/Max/Median + Anzahl NA

fertigung |>
  summarise(
    n     = n(),
    mitte = mean(masse_mm, na.rm = TRUE),     # Lagemaß
    streu = sd(masse_mm, na.rm = TRUE)        # Streumaß
  )

fertigung |>                                   # Ausschuss zählen
  mutate(ausser = abs(masse_mm - 10) > 0.15) |>
  summarise(anteil = mean(ausser, na.rm = TRUE))
